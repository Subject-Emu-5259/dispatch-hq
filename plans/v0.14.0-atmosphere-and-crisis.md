# v0.14.0 — Atmosphere & Crisis Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use `superpowers:subagent-driven-development` (recommended) or `superpowers:executing-plans` to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Make the city feel alive: a day/night cycle, weather that changes gameplay, richer synthesized audio, and city-wide disaster events that spawn waves of chained missions.

**Architecture:** A virtual game clock lives in the store (accelerated, ticks with the sim loop) and drives map styling + spawn modifiers. The existing `RandomEvents` weather state becomes gameplay-active via a `getActiveWeather()` helper consumed by `spawn.ts`. Sound stays synthesized Web Audio (`utils/sound.ts`), extended with new cue types + a `musicEnabled` ambient loop. Disasters are a new `lib/disasters.ts` module that uses `buildMission` (with v0.13.1 chain fields) to spawn timed mission waves, tracked by a store `activeDisaster` object and surfaced by a `DisasterBanner` component.

**Tech Stack:** TypeScript, Zustand, MapLibre GL, Web Audio API, Tailwind CSS, framer-motion, vitest.

## Global Constraints

- All mission construction through `apps/web/src/lib/spawn.ts` `buildMission()`.
- Weather state already exists in `apps/web/src/components/RandomEvents.tsx` — extend, don't replace.
- Settings via `apps/web/src/utils/settings.ts` (server-persisted).
- UI: dark-first Tailwind, lucide-react icons.
- Validation: `pnpm --filter @dispatch-hq/web exec tsc --noEmit`, `pnpm build`, targeted vitest runs.

```ts
export const NIGHT_SPAWN_MODIFIER = 0.7;
export const MINUTES_PER_GAME_HOUR = 30; // 1 real min = 30 game min

export function hourOfDay(gameMinutes: number): number {
  return Math.floor(gameMinutes / MINUTES_PER_GAME_HOUR) % 24;
}
export function isNight(gameMinutes: number): boolean {
  const h = hourOfDay(gameMinutes);
  return h < 6 || h >= 20;
}
export function clockLabel(gameMinutes: number): string {
  const total = Math.floor(gameMinutes / MINUTES_PER_GAME_HOUR) % 24;
  const h = total % 12 === 0 ? 12 : total % 12;
  const m = Math.floor(gameMinutes % MINUTES_PER_GAME_HOUR);
  return `${String(h).padStart(2, '0')}:${String(m).padStart(2, '0')}`;
}
```

```ts
import type { Mission, ServiceType } from '@dispatch-hq/shared';
import type { CityZone } from '@/lib/patrol/patrolZones';

export interface DisasterDefinition { /* as above */ }
export interface ActiveDisaster { /* as above */ }

export const DISASTER_DEFINITIONS: DisasterDefinition[] = [
  { id: 'tornado', name: 'Tornado Warning', icon: '🌪️', durationSec: 180, zoneDirection: 'central',
    templates: [ { templateId: 'room_fire', service: 'fire' }, { templateId: 'building_collapse', service: 'fire' }, { templateId: 'car_crash_injury', service: 'ems' } ],
    completionBonus: { credits: 1500, xp: 400 } },
  { id: 'flood', name: 'Flash Flood', icon: '🌊', durationSec: 240, zoneDirection: 'south',
    templates: [ { templateId: 'car_crash_injury', service: 'ems' }, { templateId: 'medical_call', service: 'ems' }, { templateId: 'fall_rescue', service: 'ems' } ],
    completionBonus: { credits: 1200, xp: 350 } },
  { id: 'wildfire', name: 'Wildfire Outbreak', icon: '🔥', durationSec: 300, zoneDirection: 'north',
    templates: [ { templateId: 'wildfire', service: 'fire' }, { templateId: 'room_fire', service: 'fire' }, { templateId: 'medical_call', service: 'ems' } ],
    completionBonus: { credits: 1800, xp: 500 } },
];

const DISASTER_COOLDOWN_MS = 10 * 60 * 1000;
const lastTriggerRef = { at: 0 };

export function maybeTriggerDisaster(level: number, active: ActiveDisaster | null): DisasterDefinition | null {
  if (active) return null;
  if (Date.now() - lastTriggerRef.at < DISASTER_COOLDOWN_MS) return null;
  const chance = Math.min(0.25, level * 0.02); // ~2%/level, cap 25%
  if (Math.random() > chance) return null;
  lastTriggerRef.at = Date.now();
  return DISASTER_DEFINITIONS[Math.floor(Math.random() * DISASTER_DEFINITIONS.length)];
}

export function pickDisasterZone(def: DisasterDefinition, zones: CityZone[]): CityZone | null {
  return zones.find((z) => z.direction === def.zoneDirection && z.covered !== false) ?? zones[0] ?? null;
}

export function disasterProgress(d: ActiveDisaster): number {
  return Math.max(0, Math.min(1, 1 - (d.endsAt - Date.now()) / (d.endsAt - d.startedAt)));
}
```
